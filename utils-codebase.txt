from __future__ import annotations

import numpy as np
import librosa
from typing import List, Optional, Any
from BeatNet.BeatNet import BeatNet

from audio_analysis.analysis_methods import save_in_downloads_as

from models.audio_analysis_model import AudioAnalysis

class BeatnetAnalyzer:
    """
    Analyze a local audio file using BeatNet (offline mode).

    This implementation attempts to be tolerant to different BeatNet return
    formats. It loads the audio (to get sr for conversions) then calls
    BeatNet.process(...) and extracts beat times (seconds) and optional tempo.
    """

    def analyze(self, audio_path: str) -> AudioAnalysis:
        # load audio to obtain sample rate for possible conversions
        y, sr = librosa.load(audio_path, sr=None, mono=True)
        if y.size == 0:
            raise ValueError(f"Unable to decode audio from {audio_path}")

        try:
            from BeatNet.BeatNet import BeatNet # type: ignore
        except Exception as exc:
            raise ImportError(
                "BeatNet library is required for BeatnetAnalyzer. Install it (e.g. `pip install BeatNet`) "
                "or adapt this module."
            ) from exc

        estimator = BeatNet(1, mode="offline", inference_model="DBN", plot=[], thread=False)
        output = estimator.process(audio_path)
        save_in_downloads_as(output, "beatnet_output")

        beats = [(float(t), int(p)) for t, p in output]

        
        bpm_candidates: List[float] = []
        
        beat_times = [float(b[0]) for b in sorted(beats, key=lambda x: x[0])]
        if len(beat_times) == 0 and not bpm_candidates:
            raise RuntimeError("BeatNet produced no beats for the given audio")        
        
        beat_intervals = [
            float(beat_times[i + 1] - beat_times[i])
            for i in range(len(beat_times) - 1)
        ]
        
        filtered_intervals = []
        for interval in beat_intervals:
            if not interval == 0:
                filtered_intervals.append(interval)

        processed_intervals = self._remove_anomalies(filtered_intervals)
        if processed_intervals:
            calculated_bpm = 60.0 / float(np.mean(processed_intervals))
            bpm_candidates.append(calculated_bpm)
        if filtered_intervals:
            alternative_bpm = 60.0 / float(np.median(filtered_intervals))
            bpm_candidates.append(alternative_bpm)

        final_bpm_estimate = round(float(np.mean(bpm_candidates))) if bpm_candidates else None
        # print(f"Beatnet BPM: {final_bpm_estimate}")
        # print(f"Beatnet confidence: {self._confidence_for_intervals(filtered_intervals)}")

        return AudioAnalysis(
            bpm=final_bpm_estimate,
            bpm_confidence=self._confidence_for_intervals(filtered_intervals),
            beat_intervals=filtered_intervals,
            beat_timestamps=beats,
        )

    @staticmethod
    def _confidence_for_intervals(intervals: List[float]) -> Optional[int]:
        if len(intervals) < 2:
            return None
        mean_interval = float(np.mean(intervals))
        if mean_interval <= 0:
            return None
        std_interval = float(np.std(intervals, ddof=0))
        coefficient_of_variation = int(std_interval / mean_interval * 1000)
        confidence = 1000 - coefficient_of_variation
        return max(0, min(1000, confidence))

    @staticmethod
    def _remove_anomalies(intervals: List[float]) -> List[float]:
        if not intervals:
            return []
        std_interval = float(np.std(intervals, ddof=0))
        mean_interval = float(np.mean(intervals))
        if mean_interval <= 0:
            return []
        processed_intervals: List[float] = []
        for value in intervals:
            # keep value if within +/- 0.3 * std of mean
            if (mean_interval - 0.3 * std_interval) <= value <= (mean_interval + 0.3 * std_interval):
                processed_intervals.append(float(value))
        return processed_intervals

from __future__ import annotations

import numpy as np
import librosa
from typing import List, Optional

from models.audio_analysis_model import AudioAnalysis
from audio_analysis.analysis_methods import save_in_downloads_as


class LibrosaAnalyzer:
    """
    Analyze a local audio file using librosa beat tracking.

    The BPM confidence is an application-level metric based on beat-interval
    regularity; this is not a librosa-provided confidence score.
    """

    def analyze(self, audio_path: str) -> AudioAnalysis:
        y, sr = librosa.load(audio_path, sr=None, mono=True)
        if y.size == 0:
            raise ValueError(f"Unable to decode audio from {audio_path}")

        tempo, beat_frames = librosa.beat.beat_track(y=y, sr=sr)
        bpm = []
        if isinstance(tempo, float):
            bpm.append(tempo)
        else:
            for temp in tempo:
                bpm.append(round(temp))

        beat_times = librosa.frames_to_time(beat_frames, sr=sr)
        save_in_downloads_as(beat_times, "librosa_beat_times")
        beats = [float(value) for value in beat_times]
        beat_intervals = [
            float(beats[i + 1] - beats[i]) for i in range(len(beats) - 1)
        ]
        processed_intervals = self._remove_anomalies(beat_intervals)
        if processed_intervals:
            calculated_bpm = 60 / np.mean(processed_intervals)
            alternative_bpm = 60 / np.median(beat_intervals)
        elif beat_intervals:
            calculated_bpm = 60 / np.mean(beat_intervals)
            alternative_bpm = 60 / np.median(beat_intervals)
        else:
            calculated_bpm = 0
            alternative_bpm = 0
        
        bpm.append(calculated_bpm)
        bpm.append(alternative_bpm)
        final_bpm_estimate = int(round(np.nan_to_num(np.nanmean(bpm))))
        #print(f"Librosa BPM: {final_bpm_estimate}")
        #print(f"Librosa confidence: {self._confidence_for_intervals(beat_intervals)}")
        

        return AudioAnalysis(
            bpm=final_bpm_estimate,
            bpm_confidence=self._confidence_for_intervals(beat_intervals),
            beat_intervals=beat_intervals,
            beat_timestamps=beats,
        )

    @staticmethod
    def _confidence_for_intervals(intervals: List[float]) -> Optional[int]:
        if len(intervals) < 2:
            return None

        mean_interval = float(np.mean(intervals))
        if mean_interval <= 0:
            return None

        std_interval = float(np.std(intervals, ddof=0))
        coefficient_of_variation = int(std_interval / mean_interval * 1000)
        confidence = 1000 - coefficient_of_variation
        return max(0, min(1000, confidence))

    @staticmethod
    def _remove_anomalies(intervals: List[float]) -> List[float]:
        if not intervals:
            return []
        std_interval = float(np.std(intervals, ddof=0))
        mean_interval = float(np.mean(intervals))
        if mean_interval <= 0:
            return None
        processed_intervals = []
        for i in range(len(intervals)):
            value = intervals[i]
            if not value > mean_interval + 0.3 * std_interval or value < mean_interval - 0.3 * std_interval:
                processed_intervals.append(value)
        return processed_intervals

from __future__ import annotations

import logging
import os
import shutil
import tempfile
import urllib.request
from contextlib import contextmanager
from typing import Iterator, Optional, List
import numpy as np
import math

from database.songs import fetch_unanalyzed_songs, update_song_analysis, get_song_by_id, fetch_songs
from .librosa_analyzer import LibrosaAnalyzer
from .beatnet_analyzer import BeatnetAnalyzer
from .start_time_analyzer import StartEndTimeAnalyzer
from models.song_model import Song
from models.audio_analysis_model import AudioAnalysis

logger = logging.getLogger(__name__)


class AudioAnalysisService:
    def __init__(self, db_client) -> None:
        self.db_client = db_client
        self.librosa_analyzer = LibrosaAnalyzer()
        self.beatnet_analyzer = BeatnetAnalyzer()
        self.start_end_time_analyzer = StartEndTimeAnalyzer()

    def analyze(self, limit: int = 50, force: bool = False, song_id: Optional[int] = None) -> None:
        if song_id is not None and song_id > 0:
            self.analyze_by_id(song_id)
        else:
            self.analyze_batch(limit=limit, force=force)

    def analyze_batch(self, limit: int = 50, force: bool = False) -> None:
        if force:
            songs = fetch_songs(self.db_client, limit=limit)
        else:
            songs = fetch_unanalyzed_songs(self.db_client, limit=limit)
        if not songs:
            print("No songs found for analysis.")
            return

        for index, song in enumerate(songs, start=1):
            print(f"Analyzing {index}/{len(songs)}: {song.title}")
            self.analyze_song(song)

    def analyze_by_id(self, id: int) -> None:
        song = get_song_by_id(id, self.db_client)
        print(f"Analyzing {song.title}")
        self.analyze_song(song)


    def analyze_song(self, song: Song) -> None:
        try:
            with self._prepare_local_audio(song) as audio_path:
                librosa_analysis = self.librosa_analyzer.analyze(audio_path)
                beatnet_analysis = self.beatnet_analyzer.analyze(audio_path)
                start_end_time_analysis = self.start_end_time_analyzer.analyze(song, beatnet_analysis=beatnet_analysis, librosa_analysis=librosa_analysis)
                final_analysis: AudioAnalysis = self._merge_analysis(librosa_analysis, beatnet_analysis, start_end_time_analysis)

            update_song_analysis(self.db_client, song.id, final_analysis)

            print(f"  BPM: {final_analysis.bpm}")
            print(
                f"  Confidence: "
                f"{final_analysis.bpm_confidence if final_analysis.bpm_confidence is not None else 'None'}"
            )
            print(f"  Start time: {final_analysis.start_time}")
            print(f"  End time: {final_analysis.end_time}")
        except Exception as err:
            logger.exception("Failed to analyze song %s (%s)", song.title, song.id)
            print(f"  FAILED: {err}")


    @contextmanager
    def _prepare_local_audio(self, song: Song) -> Iterator[str]:
        if song.id is None:
            raise ValueError("Song has no database id")

        source_url = song.download_url or song.audio_url
        if not source_url:
            raise ValueError("Song has no audio_url or download_url")

        if source_url.startswith("file://"):
            local_path = source_url[7:]
            cleanup = False
        elif os.path.exists(source_url):
            local_path = source_url
            cleanup = False
        else:
            local_path = self._download_remote_audio(source_url)
            cleanup = True

        try:
            yield local_path
        finally:
            if cleanup and os.path.exists(local_path):
                try:
                    os.remove(local_path)
                except OSError:
                    logger.warning("Unable to remove temporary audio file %s", local_path)

    def _download_remote_audio(self, url: str) -> str:
        suffix = os.path.splitext(url.split("?", 1)[0])[1] or ".mp3"
        fd, local_path = tempfile.mkstemp(suffix=suffix)
        os.close(fd)

        request = urllib.request.Request(
            url, headers={"User-Agent": "zwift-music/librosa-analyzer"}
        )
        with urllib.request.urlopen(request, timeout=60) as remote_stream, open(
            local_path, "wb"
        ) as local_file:
            shutil.copyfileobj(remote_stream, local_file)


        return local_path

    def _calculate_relation_estimate(self, relation: float, tolerance: float) -> float:
        if 2-tolerance < relation and relation < 2+tolerance:
            relation_estimate = 2
        elif 3-tolerance < relation and relation < 3+tolerance:
            relation_estimate = 3
        elif 4-tolerance < relation and relation < 4+tolerance:
            relation_estimate = 4
        elif 1/2 - tolerance < relation and relation < 1/2 + tolerance:
            relation_estimate = 0.5
        elif 1/3-tolerance < relation and relation < 1/3+tolerance:
            relation_estimate = 1/3
        elif 1/4-tolerance < relation and relation < 1/4+tolerance:
            relation_estimate = 1/4
        else:
            relation_estimate = 1
        return relation_estimate

    def _normalize_by_relation_estimate(self, relation_estimate: float, librosa_analysis: AudioAnalysis, beatnet_analysis: AudioAnalysis):
        beatnet_analysis_bpm = beatnet_analysis.bpm / relation_estimate
        librosa_analysis_bpm = librosa_analysis.bpm
        if relation_estimate > 1:
            k = int(relation_estimate)
            beatnet_analysis_intervals = beatnet_analysis.beat_intervals[::k]
            librosa_analysis_intervals = librosa_analysis.beat_intervals
            beatnet_analysis_timestamps = [t[0] for t in beatnet_analysis.beat_timestamps[::k]]
            librosa_analysis_timestamps = librosa_analysis.beat_timestamps
        else:
            k = int(1 / relation_estimate)
            beatnet_analysis_intervals = beatnet_analysis.beat_intervals
            librosa_analysis_intervals = librosa_analysis.beat_intervals[::k]
            beatnet_analysis_timestamps = [t[0] for t in beatnet_analysis.beat_timestamps]
            librosa_analysis_timestamps = librosa_analysis.beat_timestamps[::k]
        return librosa_analysis_bpm, beatnet_analysis_bpm, librosa_analysis_intervals, beatnet_analysis_intervals, librosa_analysis_timestamps, beatnet_analysis_timestamps
    
    def _calculate_final_bpm_and_confidence(self, librosa_bpm: float, beatnet_bpm: float, librosa_confidence: float, beatnet_confidence: float) -> int:
        if librosa_confidence is None or beatnet_confidence is None:
            if librosa_confidence is None:
                if beatnet_confidence is None:
                    final_confidence = 0
                    final_bpm = (librosa_bpm + beatnet_bpm) / 2
                else:
                    final_confidence = beatnet_confidence
                    final_bpm = beatnet_bpm
            else:
                final_confidence = librosa_confidence
                final_bpm = librosa_bpm
        elif abs(librosa_confidence - beatnet_confidence) > 100:
            if librosa_confidence > beatnet_confidence:
                final_bpm = librosa_bpm
                final_confidence = librosa_confidence - 100
            else:
                final_bpm = beatnet_bpm
                final_confidence = beatnet_confidence - 100
        elif abs(librosa_bpm - beatnet_bpm) > 15:
            final_bpm = beatnet_bpm
            final_confidence = beatnet_confidence - 100
        else:
            final_bpm = (librosa_bpm * librosa_confidence + beatnet_bpm  * beatnet_confidence) / (librosa_confidence + beatnet_confidence)
            bpm_diff = abs(librosa_bpm - beatnet_bpm)
            matchCoefficient = -1 * math.pow(bpm_diff - 5, 3)
            final_confidence = (librosa_confidence if beatnet_confidence == 0 else ((librosa_confidence + beatnet_confidence) / 2)) + matchCoefficient
        if final_confidence > 1000:
            final_confidence = 1000
        elif final_confidence < 0:
            final_confidence = 0
        if final_bpm > 160:
            final_bpm /= 2
        elif final_bpm < 40:
            final_bpm *= 2
        return int(round(final_bpm)), int(round(final_confidence))

    def _merge_intervals(self, librosa_intervals: List[float], beatnet_intervals: List[float]) -> List[float]:
        final_intervals = []
        for i in range(max(len(librosa_intervals), len(beatnet_intervals))):
            if i < len(librosa_intervals):
                librosa_interval = librosa_intervals[i]
            else:
                librosa_interval = None
            if i < len(beatnet_intervals):
                beatnet_interval = beatnet_intervals[i]
            else:
                beatnet_interval = None
            if beatnet_interval is None:
                final_intervals.append(librosa_interval)
                continue
            elif librosa_interval is None:
                final_intervals.append(beatnet_interval)
                continue
            final_intervals.append(librosa_interval + beatnet_interval / 2)
        return final_intervals
        

    def _merge_analysis(self, librosa_analysis: AudioAnalysis, beatnet_analysis: AudioAnalysis, start_end_time_analysis: AudioAnalysis) -> AudioAnalysis:
        if librosa_analysis.bpm == 0 or beatnet_analysis.bpm == 0:
            relation = 1
        else:
            relation = beatnet_analysis.bpm / librosa_analysis.bpm
        tolerance = 0.15
        relation_estimate = self._calculate_relation_estimate(relation, tolerance)
        librosa_analysis_bpm, beatnet_analysis_bpm, librosa_analysis_intervals, beatnet_analysis_intervals, librosa_analysis_timestamps, beatnet_analysis_timestamps = self._normalize_by_relation_estimate(relation_estimate, librosa_analysis, beatnet_analysis)
        final_bpm, final_confidence = self._calculate_final_bpm_and_confidence(librosa_analysis_bpm, beatnet_analysis_bpm, librosa_analysis.bpm_confidence, beatnet_analysis.bpm_confidence)
        final_intervals = self._merge_intervals(librosa_analysis_intervals, beatnet_analysis_intervals)
        final_timestamps = self._merge_intervals(librosa_analysis_timestamps, beatnet_analysis_timestamps)
        final_analysis: AudioAnalysis = AudioAnalysis(
        bpm= final_bpm,
        bpm_confidence= final_confidence,
        beat_intervals= final_intervals,
        beat_timestamps=final_timestamps,
        start_time=start_end_time_analysis.start_time,
        end_time=start_end_time_analysis.end_time
        )
        return final_analysis

from bisect import bisect_right
from typing import Any, List, Tuple, Union
from models.song_model import Song
from models.audio_analysis_model import AudioAnalysis

class StartEndTimeAnalyzer:

    def __init__(self):
        pass


    def _avg(self, values: List[tuple]|List[int]) -> float:
        """
        Returns weighted average of first values in list of tuples as a 1 point float
        """
        avg_value = 0
        total_weight = 0
        for v in values:
            if isinstance(v, int):
                avg_value += v
                total_weight += 1
            else:
                avg_value += v[0] * v[1]
                total_weight += v[1]
        return round(avg_value / total_weight if total_weight else 0, 2)

    def _closest_lower(self, sorted_list: List[Any], value: float) -> float:
        """
        Returns the closest lower value from a sorted list
        """
        if isinstance(sorted_list[0], tuple):
            sorted_list = [v[0] for v in sorted_list]
        i = bisect_right(sorted_list, value)
        return round(sorted_list[i - 1], 2) if i > 0 else None

    def _wave_to_diff(self, wave: int) -> float:
        """
        Converts waveform value to a difference value
        """
        if wave < 46.875:
            return -5/(wave-50)
        elif wave < 68.125:
            return 8/25*(wave-(335/8))
        else:
            return (-5/(wave-65)) + 10


    def _create_blocks(self, from_list: List[tuple], max_diff: float, compare: bool = False) -> List[tuple]:
        """
        Groups values into blocks based on max_diff
        """
        blocks = []
        temp = set()
        for idx, val in enumerate(from_list):
            if not temp:
                temp.add(val)
                continue
            avg = self._avg(temp)
            if abs(val[0] - avg) > max_diff:
                if compare:
                    if abs(val[0] - (from_list[idx + 1][0] if idx + 1 < len(from_list) else val[0])) > abs(val[0] - avg):
                        temp.add(val)
                        continue
                blocks.append((avg, round(from_list[idx - 1][1], 2)))
                temp.clear()
            temp.add(val)
        blocks.append((self._avg(temp), round(val[1], 2)))
        temp.clear()
        return blocks

    def _group_waveforms(self, song: Song, group_size: int) -> Tuple[List[tuple], float]:
        grouped = []
        temp = set()
        i = 0
        waveforms = len(song.waveform)
        grouped_length = song.duration/waveforms*group_size
        print(f"Grouping waveforms into groups of {group_size} (approximately {round(grouped_length, 2)} seconds), total waveforms: {waveforms}, total duration: {song.duration} seconds")
        for val in song.waveform:
            i += 1
            temp.add(val)
            if i % group_size == 0 or i == waveforms-1:
                grouped.append((self._avg(temp), round(grouped_length/group_size*i, 2)))
                temp.clear()
        return grouped, grouped_length

    def _zoom_in(self, input_list: List[Any], output_list: List[Any], input_interval_length: float, minus: int, plus: int) -> List[Any]:
        # Implementation for zooming in on specific intervals
        index = int(round(input_list[0][1]/input_interval_length)) - 1
        candidates = []
        max_diff = [0, 0]
        alt_max_diff = [0, 0]
        lower_bound = max(0, index - minus)
        upper_bound = min(len(output_list) - 1, index + plus)
        for idx, val in enumerate(output_list[lower_bound:upper_bound + 1]):
            if isinstance(val, int):
                if lower_bound > 0:
                    diff = round(abs(val - output_list[lower_bound - 1:upper_bound][idx]), 2)
                else:
                    if idx > 0:
                        diff = round(abs(val - output_list[lower_bound:upper_bound + 1][idx - 1]), 2)
                    else:
                        diff = 0
                val = (val, lower_bound + idx)
            else:
                if lower_bound > 0:
                    diff = round(abs(val[0] - output_list[lower_bound - 1:upper_bound][idx][0]), 2)
                else:
                    if idx > 0:
                        diff = round(abs(val[0] - output_list[lower_bound:upper_bound + 1][idx - 1][0]), 2)
                    else:
                        diff = 0
            candidates.append(val)
            if diff > max_diff[0] + self._wave_to_diff(val[0]):
                max_diff = [diff, idx]
            if diff > alt_max_diff[0]:
                alt_max_diff = [diff, idx]
        if max_diff is None or max_diff[0] == 0:
            max_diff = alt_max_diff
        if max_diff[1] - 1 >= 0:
            chosen = [candidates[max_diff[1] - 1], candidates[max_diff[1]]]
        else:
            chosen = [candidates[max_diff[1]]]
        return chosen

    def zoom_out_phase(self, song: Song) -> Tuple[List[tuple], List[tuple], float]:
        # Groups waveforms into groups of 10 (approximately 2 seconds) and for each group takes the average waveform
        values_grouped, grouped_length = self._group_waveforms(song, 10)
        # Goes through grouped values and compares each value to the average of the current forming group, when too distinct, adds current forming group to new list and starts forming new group
        small_blocks = self._create_blocks(values_grouped, 5, compare=False)   #5
        medium_blocks = self._create_blocks(small_blocks, 10, compare=True)
        return medium_blocks, values_grouped, grouped_length

    def zoom_in_phase(self, song: Song, medium_blocks: List[tuple], values_grouped: List[tuple], grouped_length: float, beatnet_analysis: AudioAnalysis, librosa_analysis: AudioAnalysis) -> List[tuple]:
        grouped_chosen = self._zoom_in(medium_blocks, values_grouped, grouped_length, 2, 2)
        wave_length = float(song.duration/len(song.waveform))
        wave_chosen = self._zoom_in(grouped_chosen, song.waveform, wave_length, 9, 10)
        wave_chosen_index = int(wave_chosen[0][1])
        waveform_based_start_time = round(wave_chosen_index * wave_length, 2)
        used_analysis = beatnet_analysis if beatnet_analysis.bpm_confidence and (not librosa_analysis.bpm_confidence or beatnet_analysis.bpm_confidence > librosa_analysis.bpm_confidence) else librosa_analysis
        if used_analysis is beatnet_analysis:
            timestamps = [t[0] for t in beatnet_analysis.beat_timestamps]
        else:
            timestamps = librosa_analysis.beat_timestamps
        start_time = self._closest_lower(timestamps, waveform_based_start_time)
        if start_time is None:
            start_time = waveform_based_start_time
        return start_time

    def analyze(self, song: Song, beatnet_analysis: AudioAnalysis, librosa_analysis: AudioAnalysis) -> AudioAnalysis:
        """
        Estimates best starting time in miliseconds for playing the song
        Calculates based on waveforms of the song
        """
        if song.duration == 0 or not song.waveform:
            return AudioAnalysis(
                bpm=None,
                bpm_confidence=None,
                beat_intervals=None,
                beat_timestamps=None,
                start_time=0,
                end_time=0
            )

        medium_blocks, values_grouped, grouped_length = self.zoom_out_phase(song)
        start_time = self.zoom_in_phase(song, medium_blocks, values_grouped, grouped_length, beatnet_analysis, librosa_analysis)

        reversed_medium_blocks = list(reversed(medium_blocks))

def _fetch_page(
    session: requests.Session,
    api_key: str,
    endpoint: str,
    *,
    limit: int,
    offset: int = 0,
    query: Optional[str] = None,
) -> List[Dict[str, Any]]:
    params = {
        "api_key": api_key,
        "app_name": "vibe-and-ride-utils",
        "limit": min(limit, PAGE_SIZE),
        "offset": offset,
    }
    if query:
        params["query"] = query

    response = session.get(
        f"{AUDIUS_API}/{endpoint}",
        params=params,
        headers={"x-api-key": api_key},
        timeout=30,
    )
    response.raise_for_status()

    payload = response.json()
    data = payload.get("data", payload)

    if not isinstance(data, list):
        raise RuntimeError("Audius response did not contain a track list")

    return [item for item in data if isinstance(item, dict)]


def fetch_audius_tracks(
    api_key: str,
    limit: int,
    queries: Iterable[str] = (),
) -> List[Dict[str, Any]]:
    """Fetch trending tracks and optional workout-oriented searches.

    Requests use the maximum page size and one Session to minimize overhead.
    Results are deduplicated locally before license filtering.
    """
    if limit <= 0:
        return []

    queries = [query.strip() for query in queries if query.strip()]
    tracks: Dict[str, Dict[str, Any]] = {}

    with requests.Session() as session:
        endpoints = [("tracks/trending", None)]
        endpoints.extend(("tracks/search", query) for query in queries)

        for endpoint, query in endpoints:
            offset = 0
            while len(tracks) < limit:
                page = _fetch_page(
                    session,
                    api_key,
                    endpoint,
                    limit=min(PAGE_SIZE, limit - len(tracks)),
                    offset=offset,
                    query=query,
                )
                if not page:
                    break

                for track in page:
                    track_id = str(track.get("id", "")).strip()
                    if track_id:
                        tracks.setdefault(track_id, track)

                if len(page) < PAGE_SIZE:
                    break
                offset += len(page)

    return list(tracks.values())


def import_audius_tracks(
    limit: int,
    queries: Iterable[str] = (),
    include_unlicensed: bool = False,
    include_non_commercial: bool = False,
) -> List[Song]:
    api_key = load_audius_api_key()
    raw_tracks = fetch_audius_tracks(api_key, limit, queries)

    songs: List[Song] = []
    for track in raw_tracks:
        if not include_unlicensed and not _license_is_commercial(
            track.get("license")
        ) and not track.get("is_streamable"):
            continue

        try:
            songs.append(audius_track_to_song(track))
        except (KeyError, TypeError, ValueError) as exc:
            logger.warning("Skipping malformed Audius track: %s", exc)

        if len(songs) >= limit:
            break

    return songs

        musicinfo = {}

    tags = musicinfo.get("tags") or {}
    if not isinstance(tags, dict):
        tags = {}

    result["lang"] = musicinfo.get("lang")
    result["gender"] = musicinfo.get("gender")
    result["is_instrumental"] = (
        musicinfo.get("vocalinstrumental") == "instrumental"
    )
    result["is_electric"] = (
        musicinfo.get("acousticelectric") == "electric"
    )
    result["genre"] = _normalize_list(tags.get("genres"))
    result["instruments"] = _normalize_list(tags.get("instruments"))
    result["vartags"] = _normalize_list(tags.get("vartags"))
    result["speed"] = jamendo_speed_to_int(musicinfo.get("speed"))

    result["audio_url"] = result.get("audio")
    result["download_url"] = result.get("audiodownload")
    result["waveform"] = _str_to_dict(result.get("waveform")).get("peaks")

def _optional_str(value: Any) -> Optional[str]:
    if value is None:

    return str(value)

    if value is None:

    if value is None:

    if value is None:

def _optional_float_list(value: Any) -> Optional[List[float]]:
    if value is None:
        return None
    if isinstance(value, list):
        return [float(item) for item in value]
    return [float(item) for item in list(value)]

def _str_to_list(value: str) -> List:
    """Convert string representation of list to actual list. Handles None, empty strings, and actual lists."""
    if value is None or value == "":
        return []
    if isinstance(value, list):
        return value
    try:
        result = ast.literal_eval(value)
        return result if isinstance(result, list) else []
    except (ValueError, SyntaxError):
        return []

def _str_to_dict(value: str) -> dict:
    if value is None or value == "":
        return {}
    if isinstance(value, dict):
        return value
    try:
        return json.loads(value)
    except (ValueError, json.JSONDecodeError):
        return {}

def estimate_cadence_from_power(power: float) -> Optional[float]:
    """
    Simple documented approximation mapping power -> cadence (fallback).
    Formula (first minimal implementation):
      max(75, min(115, 70 + 30 * p))
    Clamped to [75, 115] RPM.

    Rationale: 80 RPM is a reasonable mid-point for many structured workouts.
    This is only a heuristic fallback for intervals that specify power but not cadence.
    """
    if power == 0:
        power = None
    try:
        p = float(power)
    except Exception:
        return 80 #Default cadence value if there is literally no information
    cadence = round(max(75, min(115, 70 + 30 * p)))
    return cadence

def _is_bike_sport(value: Any) -> bool:
    if isinstance(value, bool):
        return value
    if value is None:
        return False
    try:
        return int(value) == 1
    except (TypeError, ValueError):
        pass
    return str(value).strip().lower() in {"bike", "bicycle", "cycling", "true", "yes", "1"}

from __future__ import annotations

import logging
import xml.etree.ElementTree as ET
from pathlib import Path
from typing import Any, List, Optional, Union

from database.workouts import upload_workout
from database.admin_client import get_admin_supabase
from models.workout_model import Workout, Interval
admin_client = get_admin_supabase()

logger = logging.getLogger(__name__)

class WorkoutParser:
    def __init__(self, client: Any | None = None):
        self.client = client if client is not None else admin_client

    def parse(self, limit: int = 50, stop_removal: bool = False) -> None:
        parsed_paths = self.parse_workouts(limit=limit)
        if not stop_removal:
            for path in parsed_paths:
                try:
                    path.unlink()
                except Exception as e:
                    logger.error(f"Failed to remove parsed workout file {path}: {e}")
        for path in parsed_paths: print(path)

    def parse_zwo_string(self, content: str, file_name: str = "Unnamed workout") -> Workout:
        root = ET.fromstring(content)
        intervals = []
        for child in root:
            if child.tag == "author":
                author = str(child.text)
            elif child.tag == "name":
                if not str(child.text) == "Zwift Workout":
                    name = str(child.text)
                else:
                    name = file_name
            elif child.tag == "sportType":
                sport_type = str(child.text)
            elif child.tag == "description":
                description = str(child.text)
            elif child.tag == "tags":
                tags = [str(tag.text) for tag in child if tag.text]
            elif child.tag == "workout":
                for ch in child:
                    intervals.append(Interval(
                        duration=float(ch.attrib.get("Duration", 0)),
                        power=float(ch.attrib.get("Power", 0)),
                        power_low=float(ch.attrib.get("PowerLow", 0)),
                        power_high=float(ch.attrib.get("PowerHigh", 0)),
                        cadence=float(ch.attrib.get("Cadence", 0)),
                        cadence_low=float(ch.attrib.get("CadenceLow", 0)),
                        cadence_high=float(ch.attrib.get("CadenceHigh", 0)),
                        type=str(ch.tag),
                    ))
        return Workout(
            name=name,
            author=author,
            sport_type=sport_type,
            tags=tags,
            description=description,
            intervals=intervals,
        )

    def parse_zwo_file(self, path: Union[str, Path]) -> None:
        path_obj = Path(path)
        with path_obj.open("r", encoding="utf-8") as file:
            source = Path(path)
            workout = self.parse_zwo_string(file.read(), source.stem)
            print(workout)
            upload_workout(workout, client=admin_client)

    def _default_asset_dirs(self) -> Path:
        base = Path(__file__).resolve().parents[1] / "assets"
        return base / "raw_workouts"

    def parse_workouts(self, limit: int = 50, raw_dir: Optional[Union[str, Path]] = None) -> List[Path]:
        if raw_dir is None:
            raw_dir = self._default_asset_dirs()
        raw_dir = Path(raw_dir)

        parsed_paths: List[Path] = []
        for source in sorted(raw_dir.glob("*.zwo"))[:limit]:
            target_path = raw_dir / f"{source.stem}.zwo"
            self.parse_zwo_file(source)
            parsed_paths.append(target_path)
        return parsed_paths
